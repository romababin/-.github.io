  function openModal(key) {
    const data = threats[key];
    document.getElementById('modalIcon').textContent = data.icon;
    document.getElementById('modalTitle').textContent = data.title;
    document.getElementById('modalContent').innerHTML = data.content;
    
    const riskLabels = {
      high: 'Высокий риск',
      medium: 'Средний риск',
      low: 'Низкий риск'
    };
    document.getElementById('modalRisk').innerHTML = 
      `<span class="risk-badge risk-${data.risk}">${riskLabels[data.risk]}</span>`;
    
    document.getElementById('modal').classList.add('active');
    document.body.style.overflow = 'hidden';
  }

  function closeModal(e) {
    if (e && e.target.id !== 'modal' && e.target.className !== 'modal-close') return;
    document.getElementById('modal').classList.remove('active');
    document.body.style.overflow = '';
  }

  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') closeModal({target: {id: 'modal'}});
  });
